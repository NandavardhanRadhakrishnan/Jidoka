# Roster Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a roster (orgs, their email domains, people, namespaced identities) and task participants, fed by sources and by hand, with a People screen and a People section in the task drawer.

**Architecture:** Validation and types live in `src/domain/roster.ts`. SQL lives in `src/repo/roster.ts` (orgs, domains, people, identities) and `src/repo/participants.ts` (task participants). Participants store raw identities and are matched to people by a join when read, so adding someone to the roster later resolves their earlier tasks. Sources report participants on `RawItem`; the poller validates and stores them. A new Hono route module serves the client. The UI follows the Claude Design handoff.

**Tech Stack:** Bun, TypeScript, `bun:sqlite`, Hono, React, `bun:test`.

**Spec:** `docs/superpowers/specs/2026-10-06-roster-design.md` (backend). UI: `docs/superpowers/specs/2026-10-06-roster-ui-handoff.md`. Where they differ, the backend spec's "UI" section wins.

## Global Constraints

- Bun only. Run `bun test <file>` for one file, `bun test` for the suite, `bun run typecheck` must stay clean after every task.
- IDs are `crypto.randomUUID()`, timestamps are `new Date().toISOString()`.
- Tests use an in-memory DB (`openDb(":memory:")` + `migrate(db)`), no network.
- Identities are `namespace:value`; the namespace is lowercased; `email:` values are lowercased whole; other values keep their case.
- Conflict messages are exact, from `ROSTER_MESSAGES` in `src/domain/roster.ts`; the API returns them as `409 { error }`.
- Public email domains: `gmail.com, googlemail.com, outlook.com, hotmail.com, live.com, yahoo.com, icloud.com, me.com, aol.com, proton.me, protonmail.com, gmx.com`.
- Client styling: classes in `src/client/modernist.css` only, no inline `style={}` in new code. Archivo weights 400/600/800 only, radius 0. Row and ghost hovers use `color-mix(in srgb, var(--color-text) 6%, transparent)` like `.type-row`.
- No "open in new tab/window" affordances (the client will move to Tauri).
- Commit messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Work on a branch, `feat/roster`, not on `main`.

## Review Focus

- **Identities with URL metacharacters in path params** (`email:a+b@x.test` in `DELETE /api/people/:id/identities/:identity`): the client `encodeURIComponent`s, the route must remove exactly that identity. Pinned in Task 7.
- **Mixed-case identities from a source** (`Email:Bob@Acme.TEST`) must resolve to the roster's `email:bob@acme.test`. Pinned in Task 4.
- **A source repeating one person with the same role** must produce one participant row, not two. Pinned in Task 3.
- **Deleting a person** removes their hand-added rows and turns source rows carrying their identities back into unknown rows, never a dangling reference. Pinned in Task 3.
- **Connecting an extension when the roster can't take the identity** (no defaults, or owned by someone else) must still succeed. Pinned in Tasks 6 (repo) and 6 (API).

export const PARTICIPANT_ROLES = ["from", "to", "cc", "author", "assignee", "commenter"] as const;

export type ParticipantRole = (typeof PARTICIPANT_ROLES)[number];

export function isParticipantRole(value: unknown): value is ParticipantRole {
  return typeof value === "string" && (PARTICIPANT_ROLES as readonly string[]).includes(value);
}

/** A relationship group seen from the user's org — not a legal company. */
export interface Org {
  id: string;
  name: string;
  /** The user's own org. Exactly one exists; it becomes the workspace in multiplayer. */
  isHome: boolean;
  /** Hints for attributing unknown senders; never moves a person by itself. */
  domains: string[];
  createdAt: string;
  updatedAt: string;
}

export interface Person {
  id: string;
  displayName: string;
  /** Always confirmed by the user; null is "No org". */
  orgId: string | null;
  /** The user. Exactly one exists. */
  isSelf: boolean;
  /** Namespaced, oldest first: "email:bob@acme.test", "github:bobsmith". */
  identities: string[];
  createdAt: string;
  updatedAt: string;
}

export interface Roster {
  orgs: Org[];
  people: Person[];
}

/** A person on an item, as a source reports it. */
export interface SourceParticipant {
  identity: string;
  name?: string;
  role: ParticipantRole;
}

export interface ResolvedParticipant {
  id: string;
  role: ParticipantRole | "added";
  addedBy: "source" | "human";
  /** The raw identity a source reported; null on hand-added rows. */
  identity: string | null;
  /** The source's display name for this identity, if it gave one. */
  sourceName: string | null;
  /** The roster person this row resolves to; null while the identity is unknown. */
  person: {
    id: string;
    displayName: string;
    isSelf: boolean;
    identities: string[];
    org: { id: string; name: string; isHome: boolean } | null;
  } | null;
}

export const PUBLIC_EMAIL_DOMAINS: ReadonlySet<string> = new Set([
  "gmail.com",
  "googlemail.com",
  "outlook.com",
  "hotmail.com",
  "live.com",
  "yahoo.com",
  "icloud.com",
  "me.com",
  "aol.com",
  "proton.me",
  "protonmail.com",
  "gmx.com",
]);

export const ROSTER_MESSAGES = {
  malformedIdentity: "Use namespace:value, e.g. email:bob@acme.test",
  malformedDomain: "Enter a domain like acme.test",
  publicDomain: (domain: string) => `${domain} is a public email domain`,
  domainTaken: (domain: string, orgName: string) => `${domain} is already on ${orgName}`,
  identityOwned: (identity: string, personName: string) => `${identity} belongs to ${personName}`,
  identityAlreadyOn: (identity: string, personName: string) => `${identity} is already on ${personName}`,
  homeOrgDelete: "Your org can't be deleted",
  selfDelete: "You can't delete yourself from the roster",
  selfOrg: "You always belong to your org",
  sourceParticipantRemove: "People reported by the source can't be removed",
} as const;

/** A request the roster refuses; the message is shown to the user verbatim. */
export class RosterConflictError extends Error {
  override name = "RosterConflictError";
}

/** An org, person, task or participant id that doesn't exist. */
export class RosterNotFoundError extends Error {
  override name = "RosterNotFoundError";
}

const NAMESPACE = /^[a-z0-9][a-z0-9_-]*$/;
const DOMAIN = /^(?=.{1,253}$)(?:[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?\.)+[a-z]{2,63}$/;

/** Lowercases, trims, strips a leading "@". Null when it isn't a domain. */
export function normalizeDomain(raw: string): string | null {
  const domain = raw.trim().toLowerCase().replace(/^@/, "");
  return DOMAIN.test(domain) ? domain : null;
}

/** Lowercases the namespace, and the whole value for `email:`. Null when malformed. */
export function normalizeIdentity(raw: string): string | null {
  const trimmed = raw.trim();
  const colon = trimmed.indexOf(":");
  if (colon <= 0) return null;
  const namespace = trimmed.slice(0, colon).toLowerCase();
  const value = trimmed.slice(colon + 1);
  if (!NAMESPACE.test(namespace) || !value || /\s/.test(value)) return null;
  if (namespace !== "email") return `${namespace}:${value}`;
  const address = value.toLowerCase();
  const parts = address.split("@");
  if (parts.length !== 2 || !parts[0] || !DOMAIN.test(parts[1] ?? "")) return null;
  return `email:${address}`;
}

export function isPublicEmailDomain(domain: string): boolean {
  return PUBLIC_EMAIL_DOMAINS.has(domain);
}

/** "acme.test" for "email:bob@acme.test"; null for any other namespace. */
export function emailDomainOf(identity: string): string | null {
  if (!identity.startsWith("email:")) return null;
  const at = identity.lastIndexOf("@");
  return at === -1 ? null : identity.slice(at + 1);
}

interface OrgRow {
  id: string;
  name: string;
  is_home: number;
  created_at: string;
  updated_at: string;
}

interface PersonRow {
  id: string;
  display_name: string;
  org_id: string | null;
  is_self: number;
  created_at: string;
  updated_at: string;
}

const now = () => new Date().toISOString();

function toOrg(row: OrgRow, domains: string[]): Org {
  return {
    id: row.id,
    name: row.name,
    isHome: row.is_home === 1,
    domains,
    createdAt: row.created_at,
    updatedAt: row.updated_at,
  };
}

function toPerson(row: PersonRow, identities: string[]): Person {
  return {
    id: row.id,
    displayName: row.display_name,
    orgId: row.org_id,
    isSelf: row.is_self === 1,
    identities,
    createdAt: row.created_at,
    updatedAt: row.updated_at,
  };
}

function groupBy(rows: { key: string; value: string }[]): Map<string, string[]> {
  const map = new Map<string, string[]>();
  for (const row of rows) map.set(row.key, [...(map.get(row.key) ?? []), row.value]);
  return map;
}

export function listOrgs(db: Database): Org[] {
  const domains = groupBy(
    db.query("SELECT org_id AS key, domain AS value FROM org_domains ORDER BY rowid").all() as {
      key: string;
      value: string;
    }[],
  );
  const rows = db.query("SELECT * FROM orgs ORDER BY created_at, rowid").all() as OrgRow[];
  return rows.map((row) => toOrg(row, domains.get(row.id) ?? []));
}

export function getOrg(db: Database, id: string): Org | null {
  const row = db.query("SELECT * FROM orgs WHERE id = ?").get(id) as OrgRow | null;
  if (!row) return null;
  const domains = (
    db.query("SELECT domain FROM org_domains WHERE org_id = ? ORDER BY rowid").all(id) as { domain: string }[]
  ).map((r) => r.domain);
  return toOrg(row, domains);
}

function requireOrg(db: Database, id: string): Org {
  const org = getOrg(db, id);
  if (!org) throw new RosterNotFoundError(`unknown org ${id}`);
  return org;
}

export function getHomeOrg(db: Database): Org {
  const row = db.query("SELECT id FROM orgs WHERE is_home = 1").get() as { id: string } | null;
  if (!row) throw new Error("getHomeOrg: no home org; call ensureRosterDefaults first");
  return requireOrg(db, row.id);
}

export function listPeople(db: Database): Person[] {
  const identities = groupBy(
    db.query("SELECT person_id AS key, identity AS value FROM person_identities ORDER BY rowid").all() as {
      key: string;
      value: string;
    }[],
  );
  const rows = db.query("SELECT * FROM people ORDER BY created_at, rowid").all() as PersonRow[];
  return rows.map((row) => toPerson(row, identities.get(row.id) ?? []));
}

export function getPerson(db: Database, id: string): Person | null {
  const row = db.query("SELECT * FROM people WHERE id = ?").get(id) as PersonRow | null;
  if (!row) return null;
  const identities = (
    db.query("SELECT identity FROM person_identities WHERE person_id = ? ORDER BY rowid").all(id) as {
      identity: string;
    }[]
  ).map((r) => r.identity);
  return toPerson(row, identities);
}

function requirePerson(db: Database, id: string): Person {
  const person = getPerson(db, id);
  if (!person) throw new RosterNotFoundError(`unknown person ${id}`);
  return person;
}

/** The person who is the user. In multiplayer this becomes the signed-in user's person. */
export function getSelf(db: Database): Person {
  const row = db.query("SELECT id FROM people WHERE is_self = 1").get() as { id: string } | null;
  if (!row) throw new Error("getSelf: no self person; call ensureRosterDefaults first");
  return requirePerson(db, row.id);
}

export function getRoster(db: Database): Roster {
  return { orgs: listOrgs(db), people: listPeople(db) };
}

/** Creates the home org ("My org") and Me when either is missing. Safe to run on every startup. */
export function ensureRosterDefaults(db: Database): void {
  db.transaction(() => {
    let home = db.query("SELECT id FROM orgs WHERE is_home = 1").get() as { id: string } | null;
    if (!home) {
      home = { id: crypto.randomUUID() };
      const at = now();
      db.query("INSERT INTO orgs (id, name, is_home, created_at, updated_at) VALUES (?, 'My org', 1, ?, ?)").run(
        home.id,
        at,
        at,
      );
    }
    if (!db.query("SELECT id FROM people WHERE is_self = 1").get()) {
      const at = now();
      db.query(
        "INSERT INTO people (id, display_name, org_id, is_self, created_at, updated_at) VALUES (?, 'Me', ?, 1, ?, ?)",
      ).run(crypto.randomUUID(), home.id, at, at);
    }
  })();
}

function touchOrg(db: Database, id: string): void {
  db.query("UPDATE orgs SET updated_at = ? WHERE id = ?").run(now(), id);
}

export function touchPerson(db: Database, id: string): void {
  db.query("UPDATE people SET updated_at = ? WHERE id = ?").run(now(), id);
}

export function createOrg(db: Database, name: string): Org {
  const id = crypto.randomUUID();
  const at = now();
  db.query("INSERT INTO orgs (id, name, is_home, created_at, updated_at) VALUES (?, ?, 0, ?, ?)").run(
    id,
    name.trim(),
    at,
    at,
  );
  return requireOrg(db, id);
}

export function renameOrg(db: Database, id: string, name: string): Org {
  requireOrg(db, id);
  db.query("UPDATE orgs SET name = ?, updated_at = ? WHERE id = ?").run(name.trim(), now(), id);
  return requireOrg(db, id);
}

/** Domains go with the org; its people move to No org (ON DELETE SET NULL). */
export function deleteOrg(db: Database, id: string): void {
  const org = requireOrg(db, id);
  if (org.isHome) throw new RosterConflictError(M.homeOrgDelete);
  db.query("DELETE FROM orgs WHERE id = ?").run(id);
}

export function addDomain(db: Database, orgId: string, raw: string): Org {
  const org = requireOrg(db, orgId);
  const domain = normalizeDomain(raw);
  if (!domain) throw new RosterConflictError(M.malformedDomain);
  if (isPublicEmailDomain(domain)) throw new RosterConflictError(M.publicDomain(domain));
  const owner = db
    .query("SELECT o.name FROM org_domains d JOIN orgs o ON o.id = d.org_id WHERE d.domain = ?")
    .get(domain) as { name: string } | null;
  if (owner) throw new RosterConflictError(M.domainTaken(domain, owner.name));
  db.query("INSERT INTO org_domains (domain, org_id) VALUES (?, ?)").run(domain, org.id);
  touchOrg(db, org.id);
  return requireOrg(db, org.id);
}

export function removeDomain(db: Database, orgId: string, raw: string): Org {
  requireOrg(db, orgId);
  const domain = normalizeDomain(raw) ?? raw;
  db.query("DELETE FROM org_domains WHERE domain = ? AND org_id = ?").run(domain, orgId);
  touchOrg(db, orgId);
  return requireOrg(db, orgId);
}

export function createPerson(db: Database, input: { name: string; orgId: string | null }): Person {
  if (input.orgId !== null) requireOrg(db, input.orgId);
  const id = crypto.randomUUID();
  const at = now();
  db.query(
    "INSERT INTO people (id, display_name, org_id, is_self, created_at, updated_at) VALUES (?, ?, ?, 0, ?, ?)",
  ).run(id, input.name.trim(), input.orgId, at, at);
  return requirePerson(db, id);
}

export function updatePerson(
  db: Database,
  id: string,
  patch: { name?: string; orgId?: string | null },
): Person {
  const person = requirePerson(db, id);
  const orgId = patch.orgId === undefined ? person.orgId : patch.orgId;
  if (orgId !== person.orgId) {
    if (person.isSelf) throw new RosterConflictError(M.selfOrg);
    if (orgId !== null) requireOrg(db, orgId);
  }
  const name = patch.name?.trim() || person.displayName;
  db.query("UPDATE people SET display_name = ?, org_id = ?, updated_at = ? WHERE id = ?").run(name, orgId, now(), id);
  return requirePerson(db, id);
}

/** Identities go with the person; so do their hand-added task rows (cascade). */
export function deletePerson(db: Database, id: string): void {
  const person = requirePerson(db, id);
  if (person.isSelf) throw new RosterConflictError(M.selfDelete);
  db.query("DELETE FROM people WHERE id = ?").run(id);
}

export function identityOwner(db: Database, identity: string): { id: string; displayName: string } | null {
  const row = db
    .query(
      "SELECT p.id, p.display_name FROM person_identities i JOIN people p ON p.id = i.person_id WHERE i.identity = ?",
    )
    .get(identity) as { id: string; display_name: string } | null;
  return row ? { id: row.id, displayName: row.display_name } : null;
}

export function addIdentity(db: Database, personId: string, raw: string): Person {
  const person = requirePerson(db, personId);
  const identity = normalizeIdentity(raw);
  if (!identity) throw new RosterConflictError(M.malformedIdentity);
  const owner = identityOwner(db, identity);
  if (owner) {
    throw new RosterConflictError(
      owner.id === person.id
        ? M.identityAlreadyOn(identity, owner.displayName)
        : M.identityOwned(identity, owner.displayName),
    );
  }
  db.query("INSERT INTO person_identities (identity, person_id) VALUES (?, ?)").run(identity, person.id);
  touchPerson(db, person.id);
  return requirePerson(db, person.id);
}

export function removeIdentity(db: Database, personId: string, raw: string): Person {
  requirePerson(db, personId);
  const identity = normalizeIdentity(raw) ?? raw;
  db.query("DELETE FROM person_identities WHERE identity = ? AND person_id = ?").run(identity, personId);
  touchPerson(db, personId);
  return requirePerson(db, personId);
}

/** People in No org who own an email identity on exactly this domain: the
 *  "move them?" suggestion after a domain is added to an org. */
export function unaffiliatedPeopleOnDomain(db: Database, raw: string): Person[] {
  const domain = normalizeDomain(raw);
  if (!domain) return [];
  const rows = db
    .query(
      `SELECT DISTINCT p.id, p.display_name FROM people p
         JOIN person_identities i ON i.person_id = p.id
        WHERE p.org_id IS NULL
          AND i.identity LIKE 'email:%'
          AND substr(i.identity, instr(i.identity, '@') + 1) = ?
        ORDER BY p.display_name`,
    )
    .all(domain) as { id: string }[];
  return rows.map((row) => requirePerson(db, row.id));
}

/** Accepts a domain suggestion. Only people still in No org move; anyone
 *  filed elsewhere in the meantime is left alone. */
export function movePeopleToOrg(db: Database, orgId: string, personIds: string[]): Person[] {
  requireOrg(db, orgId);
  return db.transaction(() => {
    const moved: Person[] = [];
    for (const id of personIds) {
      const result = db
        .query("UPDATE people SET org_id = ?, updated_at = ? WHERE id = ? AND org_id IS NULL AND is_self = 0")
        .run(orgId, now(), id);
      if (result.changes > 0) moved.push(requirePerson(db, id));
    }
    return moved;
  })();
}

import type { Database } from "bun:sqlite";
import {
  ROSTER_MESSAGES as M,
  RosterConflictError,
  RosterNotFoundError,
  type Person,
  type ResolvedParticipant,
  type SourceParticipant,
} from "../domain/roster";
import { addIdentity, createPerson, getPerson, listOrgs, listPeople } from "./roster";
import { getTask } from "./tasks";

interface Row {
  id: string;
  identity: string | null;
  person_id: string | null;
  name: string | null;
  role: string;
  added_by: "source" | "human";
  created_at: string;
  resolved_person_id: string | null;
}

const now = () => new Date().toISOString();

function requireTask(db: Database, taskId: string): void {
  if (!getTask(db, taskId)) throw new RosterNotFoundError(`unknown task ${taskId}`);
}

/** Idempotent: a naive source that reports the same people every poll writes nothing new. */
export function upsertSourceParticipants(db: Database, taskId: string, participants: SourceParticipant[]): void {
  const insert = db.query(
    `INSERT OR IGNORE INTO task_participants (id, task_id, identity, person_id, name, role, added_by, created_at)
     VALUES (?, ?, ?, NULL, ?, ?, 'source', ?)`,
  );
  for (const p of participants) insert.run(crypto.randomUUID(), taskId, p.identity, p.name ?? null, p.role, now());
}

/** Source rows resolve through their identity, hand-added rows through their person.
 *  Resolution happens here, at read time, so people added later resolve old tasks. */
export function listParticipants(db: Database, taskId: string): ResolvedParticipant[] {
  const rows = db
    .query(
      `SELECT tp.*, COALESCE(tp.person_id, pi.person_id) AS resolved_person_id
         FROM task_participants tp
         LEFT JOIN person_identities pi ON pi.identity = tp.identity
        WHERE tp.task_id = ?
        ORDER BY tp.added_by = 'human', tp.created_at, tp.rowid`,
    )
    .all(taskId) as Row[];
  if (!rows.length) return [];

  const people = new Map(listPeople(db).map((p) => [p.id, p]));
  const orgs = new Map(listOrgs(db).map((o) => [o.id, o]));
  return rows.map((row) => {
    const person = row.resolved_person_id ? (people.get(row.resolved_person_id) ?? null) : null;
    const org = person?.orgId ? (orgs.get(person.orgId) ?? null) : null;
    return {
      id: row.id,
      role: row.role as ResolvedParticipant["role"],
      addedBy: row.added_by,
      identity: row.identity,
      sourceName: row.name,
      person: person
        ? {
            id: person.id,
            displayName: person.displayName,
            isSelf: person.isSelf,
            identities: person.identities,
            org: org ? { id: org.id, name: org.name, isHome: org.isHome } : null,
          }
        : null,
    };
  });
}

/** Puts a roster person on a task by hand — unless a row on this task already
 *  resolves to them, so adding a source's unknown sender never duplicates them. */
export function tagPerson(db: Database, taskId: string, personId: string): void {
  requireTask(db, taskId);
  if (!getPerson(db, personId)) throw new RosterNotFoundError(`unknown person ${personId}`);
  if (listParticipants(db, taskId).some((p) => p.person?.id === personId)) return;
  db.query(
    `INSERT OR IGNORE INTO task_participants (id, task_id, identity, person_id, name, role, added_by, created_at)
     VALUES (?, ?, NULL, ?, NULL, 'added', 'human', ?)`,
  ).run(crypto.randomUUID(), taskId, personId, now());
}

/** "Add to roster" and "Create" in one step. A taken identity rolls the new person back. */
export function createAndTagPerson(
  db: Database,
  taskId: string,
  input: { name: string; identity?: string; orgId?: string | null },
): Person {
  requireTask(db, taskId);
  return db.transaction(() => {
    const created = createPerson(db, { name: input.name, orgId: input.orgId ?? null });
    const person = input.identity ? addIdentity(db, created.id, input.identity) : created;
    tagPerson(db, taskId, person.id);
    return person;
  })();
}

export function removeHumanParticipant(db: Database, taskId: string, participantId: string): void {
  const row = db
    .query("SELECT added_by FROM task_participants WHERE id = ? AND task_id = ?")
    .get(participantId, taskId) as { added_by: string } | null;
  if (!row) throw new RosterNotFoundError(`unknown participant ${participantId}`);
  if (row.added_by !== "human") throw new RosterConflictError(M.sourceParticipantRemove);
  db.query("DELETE FROM task_participants WHERE id = ?").run(participantId);
}

/** For a task merge: the duplicate is about to be deleted, which would cascade its rows away. */
export function moveParticipants(db: Database, fromTaskId: string, toTaskId: string): void {
  const rows = db
    .query("SELECT identity, person_id, name, role, added_by, created_at FROM task_participants WHERE task_id = ? ORDER BY rowid")
    .all(fromTaskId) as Omit<Row, "id" | "resolved_person_id">[];
  const insert = db.query(
    `INSERT OR IGNORE INTO task_participants (id, task_id, identity, person_id, name, role, added_by, created_at)
     VALUES (?, ?, ?, NULL, ?, ?, 'source', ?)`,
  );
  for (const row of rows) {
    if (row.added_by === "human" && row.person_id) tagPerson(db, toTaskId, row.person_id);
    else if (row.identity) insert.run(crypto.randomUUID(), toTaskId, row.identity, row.name, row.role, row.created_at);
  }
}

/** The valid participants of a raw item. Sources are third-party code, so
 *  anything malformed (no namespace, unknown role) is dropped and counted —
 *  bad participant data must never fail a poll. */
export function sanitizeParticipants(raw: unknown): { valid: SourceParticipant[]; dropped: number } {
  if (!Array.isArray(raw)) return { valid: [], dropped: 0 };
  const valid: SourceParticipant[] = [];
  let dropped = 0;
  for (const entry of raw as Partial<SourceParticipant>[]) {
    const identity = typeof entry?.identity === "string" ? normalizeIdentity(entry.identity) : null;
    if (!identity || !isParticipantRole(entry.role)) {
      dropped++;
      continue;
    }
    const name = typeof entry.name === "string" && entry.name.trim() ? entry.name.trim() : undefined;
    valid.push({ identity, role: entry.role, ...(name ? { name } : {}) });
  }
  return { valid, dropped };
}

/** Files an identity an extension resolved for the user onto Me. Never throws:
 *  identity resolution must never fail a connect, so a bare (un-namespaced)
 *  value, an identity someone else owns, or a roster without defaults is
 *  logged where useful and skipped. */
export function addSelfIdentity(db: Database, raw: string): "added" | "already" | "skipped" {
  try {
    const identity = normalizeIdentity(raw);
    if (!identity) return "skipped";
    const self = getSelf(db);
    const owner = identityOwner(db, identity);
    if (owner?.id === self.id) return "already";
    if (owner) {
      console.warn(`[roster] ${identity} belongs to ${owner.displayName}; not adding it to you`);
      return "skipped";
    }
    db.query("INSERT INTO person_identities (identity, person_id) VALUES (?, ?)").run(identity, self.id);
    touchPerson(db, self.id);
    return "added";
  } catch (error) {
    console.warn("[roster] could not file your identity:", error);
    return "skipped";
  }
}

import { Hono, type Context } from "hono";
import type { Database } from "bun:sqlite";
import { RosterConflictError, RosterNotFoundError } from "../domain/roster";
import {
  addDomain,
  addIdentity,
  createOrg,
  createPerson,
  deleteOrg,
  deletePerson,
  getRoster,
  movePeopleToOrg,
  removeDomain,
  removeIdentity,
  renameOrg,
  unaffiliatedPeopleOnDomain,
  updatePerson,
} from "../repo/roster";
import { createAndTagPerson, listParticipants, removeHumanParticipant, tagPerson } from "../repo/participants";
import { getTask } from "../repo/tasks";

export interface RosterRoutesDeps {
  db: Database;
}

/** The parsed JSON object body, or {} when the client sent something else. */
async function readBody(c: Context): Promise<Record<string, unknown>> {
  try {
    const parsed: unknown = await c.req.json();
    return parsed && typeof parsed === "object" ? (parsed as Record<string, unknown>) : {};
  } catch {
    return {};
  }
}

function nonEmpty(value: unknown): string | null {
  return typeof value === "string" && value.trim() ? value.trim() : null;
}

/** Roster errors onto HTTP: conflicts are 409 with the user-facing message,
 *  unknown ids are 404. Anything else is a real bug and propagates. */
async function guarded(c: Context, fn: () => Response | Promise<Response>): Promise<Response> {
  try {
    return await fn();
  } catch (error) {
    if (error instanceof RosterConflictError) return c.json({ error: error.message }, 409);
    if (error instanceof RosterNotFoundError) return c.json({ error: error.message }, 404);
    throw error;
  }
}

export function createRosterRoutes(deps: RosterRoutesDeps): Hono {
  const { db } = deps;
  const app = new Hono();

  app.get("/api/roster", (c) => c.json(getRoster(db)));

  app.post("/api/orgs", async (c) => {
    const name = nonEmpty((await readBody(c)).name);
    if (!name) return c.json({ error: "name is required" }, 400);
    return c.json({ org: createOrg(db, name) });
  });

  app.patch("/api/orgs/:id", async (c) => {
    const name = nonEmpty((await readBody(c)).name);
    if (!name) return c.json({ error: "name is required" }, 400);
    return guarded(c, () => c.json({ org: renameOrg(db, c.req.param("id"), name) }));
  });

  app.delete("/api/orgs/:id", (c) =>
    guarded(c, () => {
      deleteOrg(db, c.req.param("id"));
      return c.json({ deleted: true });
    }),
  );

  app.post("/api/orgs/:id/domains", async (c) => {
    const domain = (await readBody(c)).domain;
    if (typeof domain !== "string") return c.json({ error: "domain is required" }, 400);
    return guarded(c, () => {
      const org = addDomain(db, c.req.param("id"), domain);
      return c.json({ org, suggestedMoves: unaffiliatedPeopleOnDomain(db, domain) });
    });
  });

  app.delete("/api/orgs/:id/domains/:domain", (c) =>
    guarded(c, () => c.json({ org: removeDomain(db, c.req.param("id"), c.req.param("domain")) })),
  );

  app.post("/api/orgs/:id/people", async (c) => {
    const personIds = (await readBody(c)).personIds;
    if (!Array.isArray(personIds) || !personIds.every((id) => typeof id === "string")) {
      return c.json({ error: "personIds must be a list of ids" }, 400);
    }
    return guarded(c, () => c.json({ moved: movePeopleToOrg(db, c.req.param("id"), personIds as string[]) }));
  });

  app.post("/api/people", async (c) => {
    const body = await readBody(c);
    const name = nonEmpty(body.name);
    if (!name) return c.json({ error: "name is required" }, 400);
    const orgId = typeof body.orgId === "string" ? body.orgId : null;
    return guarded(c, () => c.json({ person: createPerson(db, { name, orgId }) }));
  });

  app.patch("/api/people/:id", async (c) => {
    const body = await readBody(c);
    const patch: { name?: string; orgId?: string | null } = {};
    if (body.name !== undefined) {
      const name = nonEmpty(body.name);
      if (!name) return c.json({ error: "name can't be empty" }, 400);
      patch.name = name;
    }
    if (body.orgId !== undefined) {
      if (body.orgId !== null && typeof body.orgId !== "string") return c.json({ error: "orgId must be an id or null" }, 400);
      patch.orgId = body.orgId;
    }
    return guarded(c, () => c.json({ person: updatePerson(db, c.req.param("id"), patch) }));
  });

  app.delete("/api/people/:id", (c) =>
    guarded(c, () => {
      deletePerson(db, c.req.param("id"));
      return c.json({ deleted: true });
    }),
  );

  app.post("/api/people/:id/identities", async (c) => {
    const identity = (await readBody(c)).identity;
    if (typeof identity !== "string") return c.json({ error: "identity is required" }, 400);
    return guarded(c, () => c.json({ person: addIdentity(db, c.req.param("id"), identity) }));
  });

  app.delete("/api/people/:id/identities/:identity", (c) =>
    guarded(c, () => c.json({ person: removeIdentity(db, c.req.param("id"), c.req.param("identity")) })),
  );

  app.get("/api/tasks/:id/participants", (c) => {
    const taskId = c.req.param("id");
    if (!getTask(db, taskId)) return c.json({ error: `unknown task ${taskId}` }, 404);
    return c.json({ participants: listParticipants(db, taskId) });
  });

  app.post("/api/tasks/:id/participants", async (c) => {
    const taskId = c.req.param("id");
    const body = await readBody(c);
    return guarded(c, () => {
      if (typeof body.personId === "string") {
        tagPerson(db, taskId, body.personId);
      } else {
        const name = nonEmpty(body.name);
        if (!name) return c.json({ error: "personId or name is required" }, 400);
        const identity = nonEmpty(body.identity);
        createAndTagPerson(db, taskId, {
          name,
          ...(identity ? { identity } : {}),
          orgId: typeof body.orgId === "string" ? body.orgId : null,
        });
      }
      return c.json({ participants: listParticipants(db, taskId) });
    });
  });

  app.delete("/api/tasks/:id/participants/:pid", (c) =>
    guarded(c, () => {
      const taskId = c.req.param("id");
      removeHumanParticipant(db, taskId, c.req.param("pid"));
      return c.json({ participants: listParticipants(db, taskId) });
    }),
  );

  return app;
}

import {
  ROSTER_MESSAGES as M,
  emailDomainOf,
  isPublicEmailDomain,
  normalizeDomain,
  normalizeIdentity,
  type Org,
  type Person,
  type Roster,
} from "../domain/roster";

/** Your org first, then the rest alphabetically. */
export function sortOrgs(orgs: Org[]): Org[] {
  return [...orgs].sort((a, b) => (a.isHome === b.isHome ? a.name.localeCompare(b.name) : a.isHome ? -1 : 1));
}

/** The people in one org (null: No org), Me first, then alphabetical. */
export function peopleInGroup(people: Person[], orgId: string | null): Person[] {
  return people
    .filter((p) => p.orgId === orgId)
    .sort((a, b) => (a.isSelf === b.isSelf ? a.displayName.localeCompare(b.displayName) : a.isSelf ? -1 : 1));
}

export function rosterSummary(roster: Roster): string {
  const orgs = roster.orgs.length;
  const people = roster.people.length;
  return `${orgs} ${orgs === 1 ? "org" : "orgs"} · ${people} ${people === 1 ? "person" : "people"}`;
}

/** The org whose domains contain this email identity's exact domain. */
export function guessOrgForIdentity(orgs: Org[], identity: string): { org: Org | null; domain: string | null } {
  const domain = emailDomainOf(identity);
  if (!domain) return { org: null, domain: null };
  return { org: orgs.find((o) => o.domains.includes(domain)) ?? null, domain };
}

export function pickerMatches(people: Person[], query: string, exclude: Set<string>, limit = 6): Person[] {
  const q = query.trim().toLowerCase();
  return people
    .filter((p) => !exclude.has(p.id))
    .filter((p) => !q || p.displayName.toLowerCase().includes(q) || p.identities.some((i) => i.includes(q)))
    .sort((a, b) => (a.isSelf === b.isSelf ? a.displayName.localeCompare(b.displayName) : a.isSelf ? -1 : 1))
    .slice(0, limit);
}

/** "Create “name”" is offered only when nobody has exactly that name. */
export function offersCreate(people: Person[], query: string): boolean {
  const q = query.trim().toLowerCase();
  return q !== "" && !people.some((p) => p.displayName.toLowerCase() === q);
}

/** The server is authoritative; this only saves a round trip with the same messages. */
export function checkDomainLocally(roster: Roster, orgId: string, raw: string): { value: string } | { error: string } {
  const domain = normalizeDomain(raw);
  if (!domain) return { error: M.malformedDomain };
  if (isPublicEmailDomain(domain)) return { error: M.publicDomain(domain) };
  const owner = roster.orgs.find((o) => o.domains.includes(domain));
  if (owner) return { error: M.domainTaken(domain, owner.name) };
  return { value: domain };
}

export function checkIdentityLocally(
  roster: Roster,
  personId: string,
  raw: string,
): { value: string } | { error: string } {
  const identity = normalizeIdentity(raw);
  if (!identity) return { error: M.malformedIdentity };
  const owner = roster.people.find((p) => p.identities.includes(identity));
  if (owner) {
    return {
      error: owner.id === personId ? M.identityAlreadyOn(identity, owner.displayName) : M.identityOwned(identity, owner.displayName),
    };
  }
  return { value: identity };
}

export function People({ onSummaryChange }: { onSummaryChange: (summary: string) => void }) {
  const [roster, setRoster] = useState<Roster | null>(null);

/** A heading-sized input that saves on blur or Enter; an empty value reverts. */
function TitleInput({ value, ariaLabel, onSave }: { value: string; ariaLabel: string; onSave: (next: string) => Promise<void> }) {
  const [draft, setDraft] = useState(value);
  useEffect(() => setDraft(value), [value]);

  function commit() {
    const next = draft.trim();
    if (!next) return setDraft(value);
    if (next !== value) void onSave(next);
  }

  return (
    <input
      className="input title-input"
      aria-label={ariaLabel}
      value={draft}
      onChange={(e) => setDraft(e.target.value)}
      onBlur={commit}
      onKeyDown={(e) => {
        if (e.key === "Enter") e.currentTarget.blur();
      }}
    />
  );
}

/** Chips with remove buttons plus an add row: shared by domains and identities. */
function ChipEditor({
  values,
  placeholder,
  emptyText,
  check,
  onAdd,
  onRemove,
}: {
  values: string[];
  placeholder: string;
  emptyText: string;
  check: (raw: string) => { value: string } | { error: string };
  onAdd: (value: string) => Promise<void>;
  onRemove: (value: string) => Promise<void>;
}) {
  const [draft, setDraft] = useState("");
  const [error, setError] = useState<string | null>(null);

  async function add() {
    const checked = check(draft);
    if ("error" in checked) return setError(checked.error);
    try {
      await onAdd(checked.value);
      setDraft("");
    } catch (e) {
      setError(message(e));
    }
  }

  return (
    <>
      {values.length ? (
        <div className="roster-chips">
          {values.map((value) => (
            <span key={value} className="mono roster-chip">
              {value}
              <button
                className="roster-chip-remove"
                aria-label={`Remove ${value}`}
                onClick={() => void onRemove(value).catch((e) => setError(message(e)))}
              >
                <Icon name="x" size={12} />
              </button>
            </span>
          ))}
        </div>
      ) : (
        <p className="roster-note">{emptyText}</p>
      )}
      <div className="roster-add-row">
        <input
          className="input mono"
          placeholder={placeholder}
          value={draft}
          onChange={(e) => {
            setDraft(e.target.value);
            setError(null);
          }}
          onKeyDown={(e) => {
            if (e.key === "Enter") void add();
          }}
        />
        <button className="btn btn-secondary btn-sm" disabled={!draft.trim()} onClick={() => void add()}>
          Add
        </button>
      </div>
      {error && <p className="error-text">{error}</p>}
    </>
  );
}

function PersonDetail({

function OrgBadge({ org }: { org: { name: string; isHome: boolean } | null }) {
  if (!org) return <span className="org-badge org-none">no org</span>;

export function TaskPeople({ taskId }: { taskId: string }) {
  const [participants, setParticipants] = useState<ResolvedParticipant[]>([]);
  const [roster, setRoster] = useState<Roster | null>(null);
  const [picking, setPicking] = useState(false);
  const [addingFor, setAddingFor] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  async function reload() {
    const [nextParticipants, nextRoster] = await Promise.all([api.participants(taskId), api.roster()]);
    setParticipants(nextParticipants);
    setRoster(nextRoster);

  return (

function AddToRosterForm({
  taskId,
  identity,
  sourceName,
  orgs,
  onDone,
  onCancel,
}: {
  taskId: string;
  identity: string;
  sourceName: string;
  orgs: Org[];
  onDone: () => Promise<void>;
  onCancel: () => void;
}) {
  const guess = guessOrgForIdentity(orgs, identity);
  const [name, setName] = useState(sourceName);
  const [editedIdentity, setEditedIdentity] = useState(identity);
  const [orgId, setOrgId] = useState(guess.org?.id ?? "");
  const [orgTouched, setOrgTouched] = useState(false);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const org = orgs.find((o) => o.id === orgId);

  async function submit() {
    setBusy(true);
    setError(null);
    try {
      await api.addParticipant(taskId, {
        name: name.trim(),
        ...(editedIdentity.trim() ? { identity: editedIdentity.trim() } : {}),
        orgId: orgId || null,
      });
      await onDone();
    } catch (e) {
      setError(message(e));
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="participant-add-form">
      <label className="roster-field">
        <span className="roster-label">Display name</span>
        <input className="input" value={name} onChange={(e) => setName(e.target.value)} />
      </label>
      <label className="roster-field">
        <span className="roster-label">Identity</span>
        <input className="input mono" value={editedIdentity} onChange={(e) => setEditedIdentity(e.target.value)} />
      </label>
      <div className="roster-field-row">
        <SelectTrigger
          label="org"
          display={<span className="mono">{org ? org.name : "No org"}</span>}
          value={orgId}
          ariaLabel="Org"
          onChange={(value) => {
            setOrgId(value);
            setOrgTouched(true);
          }}
        >
          {sortOrgs(orgs).map((o) => (
            <option key={o.id} value={o.id}>
              {o.name}
            </option>
          ))}
          <option value="">No org</option>
        </SelectTrigger>
        {!orgTouched && guess.domain && (
          <span className="roster-note">
            {guess.org ? `Guessed from ${guess.domain}` : `No org has ${guess.domain}`}
          </span>
        )}
      </div>
      <div className="roster-actions">
        <button className="btn btn-secondary btn-sm" disabled={!name.trim() || busy} onClick={() => void submit()}>
          <Icon name="user-plus" size={13} /> Add to roster
        </button>
        <button className="btn btn-ghost btn-xs" onClick={onCancel}>
          Cancel
        </button>
      </div>
      {error && <p className="error-text">{error}</p>}
    </div>
  );
}

function PersonPicker({
  taskId,
  roster,
  participants,
  onDone,
  onCancel,
}: {
  taskId: string;
  roster: Roster;
  participants: ResolvedParticipant[];
  onDone: () => Promise<void>;
  onCancel: () => void;
}) {
  const [query, setQuery] = useState("");
  const [error, setError] = useState<string | null>(null);
  const onTask = new Set(participants.flatMap((p) => (p.person ? [p.person.id] : [])));
  const matches = pickerMatches(roster.people, query, onTask);
  const canCreate = offersCreate(roster.people, query);
  const orgById = new Map(roster.orgs.map((o) => [o.id, o]));

  async function add(input: { personId: string } | { name: string }) {
    try {
      await api.addParticipant(taskId, input);
      await onDone();
    } catch (e) {
      setError(message(e));
    }
  }

  return (
    <div className="participant-picker">
      <div className="roster-add-row">
        <input
          className="input"
          autoFocus
          placeholder="Find someone in the roster, or type a new name"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
        />
        <button className="btn btn-ghost btn-xs" onClick={onCancel}>
          Cancel
        </button>
      </div>
      {matches.map((person) => (
        <button key={person.id} className="participant-pick" onClick={() => void add({ personId: person.id })}>
          <span className="name">{person.displayName}</span>
          {person.isSelf && <span className="me-badge">me</span>}
          <OrgBadge org={person.orgId ? (orgById.get(person.orgId) ?? null) : null} />
          <span className="mono ident">{person.identities[0] ?? "no identities"}</span>
        </button>
      ))}
      {canCreate && (
        <button className="participant-pick create" onClick={() => void add({ name: query.trim() })}>
          <span className="name">Create “{query.trim()}”</span>
          <span className="roster-note">Adds them to the roster with no org and no identities.</span>
        </button>
      )}
      {!matches.length && !canCreate && <p className="roster-note">Everyone in the roster is already on this task.</p>}
      {error && <p className="error-text">{error}</p>}
    </div>
  );
}
